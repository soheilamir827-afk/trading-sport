async function fetchPrice(asset, symbol) { const status = document.getElementById( asset === 'BTCUSDT' ? 'btcStatus' : 'ethStatus' ); const sources = [ async () => { const r = await fetch( 'https://api.binance.com/api/v3/ticker/price?symbol=' + symbol ); if (!r.ok) throw new Error('Binance'); return Number((await r.json()).price); }, async () => { const coin = asset === 'BTCUSDT' ? 'bitcoin' : 'ethereum'; const r = await fetch( 'https://api.coingecko.com/api/v3/simple/price?ids=' + coin + '&vs_currencies=usd' ); if (!r.ok) throw new Error('CoinGecko'); const d = await r.json(); return Number(d[coin].usd); } ]; for (const source of sources) { try { const price = await source(); if (!Number.isFinite(price) || price <= 0) { throw new Error('Invalid price'); } state.prices[asset] = price; status.textContent = '● قیمت دریافت شد'; status.className = 'subtle green'; return; } catch (e) { // امتحان منبع بعدی } } status.textContent = 'قیمت دریافت نشد؛ اتصال شبکه را بررسی کن'; status.className = 'subtle red'; }
